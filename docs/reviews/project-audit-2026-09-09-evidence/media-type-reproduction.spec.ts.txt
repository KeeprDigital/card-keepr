import { env } from "cloudflare:workers";
import { expect, test } from "vitest";
import { requiredSourceAdapter } from "/Users/marcus/Developer/card-keepr-worktrees/project-audit-20260909/src/catalogue/adapters";
import { catalogueStore, sha256, utf8 } from "/Users/marcus/Developer/card-keepr-worktrees/project-audit-20260909/src/catalogue/shared";
import { parseSnapshot } from "/Users/marcus/Developer/card-keepr-worktrees/project-audit-20260909/src/catalogue/source-evidence/source-evidence-parsing";
import { installRuntimeSuite } from "/Users/marcus/Developer/card-keepr-worktrees/project-audit-20260909/apps/ingestion/test/runtime-helpers";
import { seedRunFixtureStatement } from "/Users/marcus/Developer/card-keepr-worktrees/project-audit-20260909/apps/ingestion/test/query-helpers/run-events";

installRuntimeSuite();

test("document response labelled image bypasses its Bandai parser and seals zero records", async () => {
  const db = env.CATALOGUE_DB;
  const run = "audit-image-response", snapshot = "audit-image-snapshot", attempt = "audit-image-attempt";
  const request = "one-piece-en:card-list", version = "one-piece-en@6";
  const url = "https://en.onepiece-cardgame.com/cardlist/?series=569116";
  const bytes = utf8("unexpected non-HTML response body");
  const key = `source-snapshots/${snapshot}`;
  await seedRunFixtureStatement(db, { id: run, state: "collecting", idempotency_key: run }).run();
  await db.prepare(`INSERT INTO source_requests
    (ingestion_run_id,request_id,sequence_number,method,url,request_headers_json,representation_fingerprint,state)
    VALUES (?,?,1,'GET',?,'{}','fixture','captured')`).bind(run, request, url).run();
  await db.prepare(`INSERT INTO source_fetch_attempts
    (id,ingestion_run_id,request_id,attempt_number,requested_at,completed_at,outcome,response_headers_json)
    VALUES (?,?,?,1,'2026-09-08T00:00:00.000Z','2026-09-08T00:00:00.000Z','success','{}')`)
    .bind(attempt, run, request).run();
  await db.prepare(`INSERT INTO source_snapshots
    (id,ingestion_run_id,request_id,fetch_attempt_id,request_method,request_url,request_headers_json,
     representation_fingerprint,response_vary_json,retrieved_at,http_status,response_headers_json,media_type,
     content_digest,content_byte_length,content_object_key,source_lineage,supported_game,game_profile_version,adapter_version)
    VALUES (?,?,?,?,'GET',?,'{}','fixture','[]','2026-09-08T00:00:00.000Z',200,'{}','image/png',?,?,?,
      'one-piece-en','one-piece','one-piece@1',?)`)
    .bind(snapshot, run, request, attempt, url, await sha256(bytes), bytes.length, key, version).run();
  await env.EVIDENCE_OBJECTS.put(key, bytes);

  // Existing adapter rejects this non-image request's media type/body.
  const adapter = requiredSourceAdapter(version);
  expect(() => adapter.parseBytes!(bytes, { url, mediaType: "image/png", requestId: request })).toThrow();

  // The new intake boundary instead skips that parser and seals empty evidence.
  const observed = await parseSnapshot(catalogueStore(db), env.EVIDENCE_OBJECTS, snapshot, version,
    { intent: "collection", idempotencyKey: run });
  expect(observed.observation_count).toBe(0);
  expect(await db.prepare("SELECT sealed FROM source_record_progress WHERE observation_set_id=?")
    .bind(observed.id).first()).toEqual({ sealed: 1 });
});
